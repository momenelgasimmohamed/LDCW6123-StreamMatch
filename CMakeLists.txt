cmake_minimum_required(VERSION 3.16)
project(StreamMatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(recommender src/recommender.cpp)
target_include_directories(recommender PUBLIC src)
add_executable(streammatch src/main.cpp)
target_link_libraries(streammatch PRIVATE recommender)
foreach(target recommender streammatch)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
